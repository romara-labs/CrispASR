cmake_minimum_required(VERSION 3.16)
if(NOT TARGET ggml)
    project(cann_weight_layout_probe LANGUAGES C CXX)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    add_subdirectory(../../../ggml ggml)
endif()
add_executable(cann-weight-layout probe.cpp)
target_link_libraries(cann-weight-layout PRIVATE ggml ggml-cpu)
target_compile_features(cann-weight-layout PRIVATE cxx_std_17)
if(TARGET ggml-cann)
    target_link_directories(cann-weight-layout PRIVATE "$<TARGET_PROPERTY:ggml-cann,LINK_DIRECTORIES>")
endif()
